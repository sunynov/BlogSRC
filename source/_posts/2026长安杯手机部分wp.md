```
rsync -rt --partial --timeout=300 --mkpath --no-perms --no-owner --no-group --omit-dir-times -e /data/app/~~eQjvGIzFe_ehHh-aONJ2wA==/invalid.lena.rsend-DQHZyQtwgRke5u50mTMZRQ==/lib/arm64/libxrsh.so /storage/emulated/0/sync/ rsync@47.89.230.121:/home/rsync/sync_phone/
```
